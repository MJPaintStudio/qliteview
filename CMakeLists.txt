cmake_minimum_required(VERSION 3.24)
project(qliteview VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Quick WebChannel)
qt_standard_project_setup(REQUIRES 6.11)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "QLiteView currently targets Windows x64.")
endif()
set(QLITEVIEW_ENGINE_BUILD "${CMAKE_SOURCE_DIR}/build/windows-gl" CACHE PATH "Native engine build directory")
set(QLITEVIEW_RUNTIME_DIR "${CMAKE_SOURCE_DIR}/build/runtime" CACHE PATH "Prepared runtime directory")
set(QT_QML_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory(src/QWebKit)
option(QLITEVIEW_BUILD_EXAMPLE "Build the interactive browser example" ON)
if(QLITEVIEW_BUILD_EXAMPLE)
    find_package(Qt6 6.11 REQUIRED COMPONENTS QuickControls2)
    add_subdirectory(examples/browser)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.11 REQUIRED COMPONENTS Test Network)
    add_subdirectory(tests)
endif()
